$env:Path = "$env:USERPROFILE\.cargo\bin;$env:Path"                                           
cargo --version
pnpm --dir apps/player tauri:dev

Add-Type -AssemblyName System.IO.Compression.FileSystem
$root = (Get-Location).Path
$zipPath = Join-Path (Split-Path $root -Parent) 'first-ivf-forward-slashes.zip'
$archive = [System.IO.Compression.ZipFile]::Open(
  $zipPath,
  [System.IO.Compression.ZipArchiveMode]::Create
)

try {
  [System.IO.Compression.ZipFileExtensions]::CreateEntryFromFile(
    $archive, (Join-Path $root 'manifest.json'), 'manifest.json'
  ) | Out-Null

  Get-ChildItem (Join-Path $root 'media') -File -Recurse | ForEach-Object {
    $entry = $_.FullName.Substring($root.Length + 1).Replace('\', '/')
    [System.IO.Compression.ZipFileExtensions]::CreateEntryFromFile(
      $archive, $_.FullName, $entry
    ) | Out-Null
  }
}
finally {
  $archive.Dispose()
}

Rename-Item $zipPath 'first-ivf-forward-slashes.ivxpkg'
